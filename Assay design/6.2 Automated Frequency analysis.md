# Automated Frequency Analysis
*This step describes how to check the tissue distribution of mutations within selected target regions.* <br>

*The analysis was performed only for sites selected for primer design. The aim was to evaluate whether mutations in the selected target site or region were more or less common in the intended target region.*

*OBS! This code gives the number of unique samples after filtering; to know the unique number of patients, a manual control has to be made for each sample where you enter each Sample ID on Cosmic to check for "Other samples linked to the same individual".*

Paste the following code:

```
import csv
import re
from pathlib import Path

# ======= Settings =======

# Original COSMIC files, one containing positive mutation data and one containing all data for the gene of interest 
POS_FILE = Path("MUC16_POS.tsv")
ALL_FILE = Path("MUC16_ALL.tsv")

# Tissue of interest, change to the tissue of interest
TARGET_TISSUE = "Endometrium"

# Mutation annotations used to define the target position
# "CDS Mutation" or "AA Mutation", change to the desired target option
MUTATION_COLUMN = "CDS Mutation"

# Analysis mode, change to the desired option
# "site"  = one exact position
# "range" = interval of positions
ANALYSIS_MODE = "site"

# Exact mutation position used when ANALYSIS_MODE = "site", change to desired intervall
TARGET_SITE = 16757

# Start and end position used when ANALYSIS_MODE = "range", change to desired intervall
RANGE_START = 18649
RANGE_END = 18670


# Column names expected in the downloaded COSMIC TSV files
COL_SAMPLE_ID = "Sample ID"
COL_TISSUE = "Primary Tissue"

# ==================================================================================================
# ======= Functions =======
#Extract the numerical mutation position from the COSMIC mutation annotation, either CDS  or AA mutation
def extract_position(value, mutation_column):

    # Ignore empty mutation annotations
    if not value:
        return None

    value = value.strip()

    # Extract position from CDS position, c.-notation
    if mutation_column == "CDS Mutation":
        match = re.search(r"c\.(\d+)", value)

    # Extract position from AA position
    elif mutation_column == "AA Mutation":
        match = re.search(r"(\d+)", value)

    else:
        raise ValueError(
            "MUTATION_COLUMN must be 'CDS Mutation' or 'AA Mutation'."
        )

    if match:
        return int(match.group(1))

    return None

==================================================================================================
# Read a COSMIC file and retain rows belonging to the selected Tissue.
def filter_by_tissue(path, target_tissue):
    
    matching_rows = []

    # Open input file as a tab-separated table.
    with path.open(
        "r",
        encoding="utf-8-sig",
        newline=""
    ) as f:

        reader = csv.DictReader(
            f,
            delimiter="\t"
        )

        # Check if the input file contains a header
        if not reader.fieldnames:
            raise ValueError(
                f"{path.name} is empty or has no header."
            )

	# Sample ID and Tissue are required for the analysis 
        required_columns = {
            COL_SAMPLE_ID,
            COL_TISSUE
        }

        missing = (
            required_columns
            - set(reader.fieldnames)
        )

        if missing:
            raise ValueError(
                f"{path.name} is missing required columns: "
                f"{', '.join(sorted(missing))}"
            )

	# Retain only rows belonging to the selected tissue
        for row in reader:

            tissue = (
                row.get(COL_TISSUE) or ""
            ).strip()

            if (
                tissue.casefold()
                == target_tissue.casefold()
            ):
                matching_rows.append(row)

    return matching_rows

# ==================================================================================================
# Filter already tissue-selected POS data by either one exact mutation position or a selected mutation interval. 
def filter_by_position(
    rows,
    mutation_column,
    analysis_mode,
    target_site=None,
    range_start=None,
    range_end=None
):
    
    # Return an empty result if no rows remain after tissue filtering
    if not rows:
        return []

    # Check that the selected mutation annotation exists in the POS dataset
    if mutation_column not in rows[0]:
        raise ValueError(
            f"Column '{mutation_column}' was not found "
            "in the POS dataset."
        )

    # An exact target position is required for site analysis
    if analysis_mode not in {"site", "range"}:
        raise ValueError(
            "ANALYSIS_MODE must be 'site' or 'range'."
        )

    # An exact target position is required for site analysis
    if analysis_mode == "site" and target_site is None:
        raise ValueError(
            "TARGET_SITE must be specified when "
            "ANALYSIS_MODE = 'site'."
        )

    # Start and end position are required for range analysis
    if analysis_mode == "range":

        if range_start is None or range_end is None:
            raise ValueError(
                "RANGE_START and RANGE_END must be specified "
                "when ANALYSIS_MODE = 'range'."
            )

        if int(range_start) > int(range_end):
            raise ValueError(
                "RANGE_START cannot be larger than RANGE_END."
            )

    matching_rows = []

    # Go through all mutation-positive rows from the selected tissue
    for row in rows:

        mutation = (
            row.get(mutation_column) or ""
        ).strip()

        position = extract_position(
            mutation,
            mutation_column
        )

        # Convert the mutation annotation to a numerical position
        if position is None:
            continue

	# Exact site analysis
        if analysis_mode == "site":

            if position == int(target_site):
                matching_rows.append(row)

	# Target region analysis
        elif analysis_mode == "range":

            if (
                int(range_start)
                <= position
                <= int(range_end)
            ):
                matching_rows.append(row)

    return matching_rows

# ==================================================================================================
# Identify unique COSMIC samples. 
# Sample ID is used as the unique sample identifier because COSMIC defines Sample ID as the unique ID of the sample in which the sample was found. 
# The same Sample ID may occur on several rows, to account for that, the use of a set ensures that each sample contributes only once to the sample count. 
def get_unique_samples(rows):

    samples = set()

    for row in rows:

        sample_id = (
            row.get(COL_SAMPLE_ID) or ""
        ).strip()

	# Ignore rows without a Sample ID
        if sample_id:
            samples.add(sample_id)

    return samples

# ==================================================================================================
# Generate a readable description of the selected target for the terminal output.
def get_target_label():

    if ANALYSIS_MODE == "site":
        return f"site {TARGET_SITE}"

    return f"range {RANGE_START}-{RANGE_END}"

# ==================================================================================================
# ======= Main analysis =======
def main():

    print("\n======================================")
    print("COSMIC mutation frequency analysis")
    print("======================================")

    target_label = get_target_label()

    # Print selected analysis settings 
    print(f"\nPrimary Tissue: {TARGET_TISSUE}")
    print(f"Mutation column: {MUTATION_COLUMN}")
    print(f"Analysis mode: {ANALYSIS_MODE}")
    print(f"Target: {target_label}")


    # Check if both COSMIC input files are avalible
    if not ALL_FILE.exists():
        raise FileNotFoundError(
            f"ALL file not found: {ALL_FILE.resolve()}"
        )

    if not POS_FILE.exists():
        raise FileNotFoundError(
            f"POS file not found: {POS_FILE.resolve()}"
        )


# ==================================================================================================
    # Filter ALL data by tissue, this represents all avalible samples from the selected tissue
    all_tissue_rows = filter_by_tissue(
        ALL_FILE,
        TARGET_TISSUE
    )

    # Filter POS data by tissue, this represents all mutation-positive samples from the selected tissue
    pos_tissue_rows = filter_by_tissue(
        POS_FILE,
        TARGET_TISSUE
    )

    # Further filter the POS data by the selected targeted site or region 
    pos_target_rows = filter_by_position(
        pos_tissue_rows,
        MUTATION_COLUMN,
        ANALYSIS_MODE,
        target_site=TARGET_SITE,
        range_start=RANGE_START,
        range_end=RANGE_END
    )

# ==================================================================================================
    # Sample ID is used to ensure that samples appearing on several rows contribute once to each count
    all_tissue_samples = get_unique_samples(
        all_tissue_rows
    )

    pos_tissue_samples = get_unique_samples(
        pos_tissue_rows
    )

    pos_target_samples = get_unique_samples(
        pos_target_rows
    )

    # Count unique samples in each group
    n_all_tissue = len(all_tissue_samples)
    n_pos_tissue = len(pos_tissue_samples)
    n_pos_target = len(pos_target_samples)

# ==================================================================================================
    # Calculate the proportion of all samples from the selected tissue that contain a mutation within the selected target site or region
    if n_all_tissue > 0:

        proportion_all = (
            n_pos_target
            / n_all_tissue
            * 100
        )

    else:
        proportion_all = 0

    # Calculate the proportion of mutation-positive samples from the selected tissue that contain a mutation within the selected target site or region 
    if n_pos_tissue > 0:

        proportion_positive = (
            n_pos_target
            / n_pos_tissue
            * 100
        )

    else:
        proportion_positive = 0

# ==================================================================================================
# ======= Output =======

    # Raw readcounts are reported for traceability, these values may be higher than the number of unique samples because one sample ID can occur on several mutation rows
    print("\n--- Raw rows ---")

    print(
        f"ALL data - rows in {TARGET_TISSUE}: "
        f"{len(all_tissue_rows)}"
    )

    print(
        f"POS data - rows in {TARGET_TISSUE}: "
        f"{len(pos_tissue_rows)}"
    )

    print(
        f"POS data - rows in {TARGET_TISSUE} "
        f"at {target_label}: "
        f"{len(pos_target_rows)}"
    )

    # Report the number of unique COSMIC sample IDs
    print("\n--- Unique samples ---")

    print(
        f"ALL data - unique samples in "
        f"{TARGET_TISSUE}: "
        f"{n_all_tissue}"
    )

    print(
        f"POS data - unique samples in "
        f"{TARGET_TISSUE}: "
        f"{n_pos_tissue}"
    )

    print(
        f"POS data - unique samples in "
        f"{TARGET_TISSUE} at {target_label}: "
        f"{n_pos_target}"
    )

    # Report both mutation frequency calculations. 
    print("\n--- Proportions ---")

    print(
        f"Target / ALL tissue samples: "
        f"{n_pos_target} / {n_all_tissue} "
        f"= {proportion_all:.2f}%"
    )

    print(
        f"Target / POS tissue samples: "
        f"{n_pos_target} / {n_pos_tissue} "
        f"= {proportion_positive:.2f}%"
    )



    # List all unique Sample IDs contributing to the selected target
    # These can be used for manual inspection in COSMIC if required
    if pos_target_samples:

        print(
            f"\n--- Sample IDs at {target_label} ---"
        )

        for sample_id in sorted(
            pos_target_samples
        ):
            print(sample_id)

    else:

        print(
            f"\nNo samples found at {target_label}."
        )


    print("\n======================================\n")


if __name__ == "__main__":
    main()
```
